cmake_minimum_required(VERSION 3.16)
project(qpainter_mbt LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets)
if(NOT DEFINED MOON_HOME)
  set(MOON_HOME "$ENV{HOME}/.moon")
endif()
add_library(qpainter_mbt STATIC stub.cxx test_support.cxx)
target_include_directories(qpainter_mbt PRIVATE "${MOON_HOME}/include")
target_link_libraries(qpainter_mbt PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets)
option(QPAINTER_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
if(QPAINTER_SANITIZE)
  target_compile_options(qpainter_mbt PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
